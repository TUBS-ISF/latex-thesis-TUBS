\ifthesis@notomath
    \IfFontExistsTF{Noto Sans Math}{%
        \thesis@debug{Setup Noto Math font}%
        \setmathfont{Noto Sans Math}%
    }{%
        \thesis@warn{The font 'Noto Sans Math' is not installed, falling back
          to 'TeX Gyre Pagella Math'. Install the Noto fonts or pass
          'notomath=false' to silence this warning}%
        \setmathfont{texgyrepagella-math.otf}%
    }
\else

    \setmathfont{texgyrepagella-math.otf}
