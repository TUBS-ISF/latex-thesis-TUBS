\def\makeproposalinclusion#1{
        \cleardoublepage\begingroup\chapter{\translate{task}}\label{ch:task}

        \translate{task-description}

        \par\@sigskip\@signarea\thesis@voidpar\endgroup\bigskip\thesis@voidpar

        \begin{refsection}
        \captionsetup{list=no}
        \input{#1}

        {%
        \defbibheading{bibliography}[\bibname]{\section*{\translate{bibliography}}}%
        \printbibliography%
        }%
        \captionsetup{list=yes}
        \end{refsection}

}

\newcommand{\makeproposaltitle}{
    {\begin{center}
        {\LARGE Proposal --- \@title}\\
        { \@thesistype's Thesis Proposal, \@author} \\
    \end{center}

    {\bfseries\ifnum\value{thesis@examiners}=1
    \translate{advisor-singular}%
    \else\ifnum\value{thesis@examiners}>1
        \translate{advisor-plural}%
        \fi\fi\smallskip}
    \ifnum\value{thesis@examiners}>0
    \begin{itemize}[leftmargin=*,nosep,itemsep=\smallskipamount]
    \@examiners
    \end{itemize}
    \fi
    \vskip.5cm
        {\bfseries\ifnum\value{thesis@supervisors}=1
    \translate{supervisor-singular}%
    \else\ifnum\value{thesis@supervisors}>1
        \translate{supervisor-plural}%
        \fi\fi\smallskip}
    \ifnum\value{thesis@supervisors}>0
    \begin{itemize}[leftmargin=*,nosep,itemsep=\smallskipamount]
    \@supervisors
    \end{itemize}
    \fi
    \vskip.5cm
    }
}

\def\makeganttchart#1{
    \ifdefstring{\thesis@type}{master-thesis}{%
        \begin{figure}
            \centering
            \begin{ganttchart}[hgrid, vgrid,expand chart=\textwidth, milestone/.append style={fill=@primary, rounded corners=3pt}, bar label font=\small, milestone label font=\footnotesize]{1}{24}
			\gantttitle{\translate{week}}{24} \\
            \gantttitlelist{1,...,24}{1} \\
            #1
            \end{ganttchart}
            \caption{\translate{ganttcap}} \label{fig:proposal:gantt}
        \end{figure}
    }{}

    \ifdefstring{\thesis@type}{bachelor-thesis}{%
        \begin{figure}
            \centering
            \begin{ganttchart}[hgrid, vgrid,expand chart=\textwidth, milestone/.append style={fill=@primary, rounded corners=3pt}, bar label font=\small, milestone label font=\footnotesize]{1}{12}
                \gantttitle{\translate{week}}{12} \\
                \gantttitlelist{1,...,12}{1} \\
                #1
            \end{ganttchart}
            \caption{\translate{ganttcap}} \label{fig:proposal:gantt}
        \end{figure}
    }{}
}
