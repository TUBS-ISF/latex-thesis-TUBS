% lockcites in the sidebar inspired by my bachelor's thesis; a citation appears
% at most once per scope, the scope is chosen by the citeMarginparScope option
\def\thesis@sidecite@scope{\csname thesis@citescope@\thesis@citeMarginparScope\endcsname}
\def\thesis@sidecite@do#1{\tiny\color{darkgray}\plaincite{#1}~\thesis@sidecite{#1}}
\def\@locksidecite{%
   \expandafter\expandafter\expandafter\xdef\expandafter\csname mcited\etry\endcsname{\thesis@sidecite@scope}%
   \expandsidenote[\z@]{\protect\thesis@sidecite@do{\etry}}%
}
\newif\if@sidecite@doreset@
\def\forcecite#1{{\@sidecite@doreset@true\sidecite{#1}\plaincite{#1}}}
\def\sidecite#1{%
   \edef\@chk{\detokenize{#1}}%
   \ifx\@chk\@empty\else
   \@for\etry:=\@chk\do{%
      \if@sidecite@doreset@
         \csundef{mcited\etry}%
      \fi
      \ifthesis@citescope@always
         \@locksidecite
      \else
      \expandafter\ifcsname mcited\etry\endcsname\xdef\@tmp{\thesis@sidecite@scope}%
         \expandafter\expandafter\expandafter\ifx\expandafter\csname mcited\etry\endcsname
            \@tmp
         \else
            \@locksidecite
         \fi
      \else
         \@locksidecite
      \fi
      \fi
   }\fi%
}
